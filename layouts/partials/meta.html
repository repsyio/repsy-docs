{{- /* Description: front matter `description` (every page in content/, content-os/ and shared/ has one), else for the generated tag and category pages a fixed sentence, else the page summary without the leading heading, else the site-level `description` param. Omitted when none is available. Also emits the Open Graph and Twitter card tags from the same title, description and canonical URL as head.html; they are left off the 404 page. */ -}} {{- $description := .Description -}} {{- if and (not $description) (in (slice "taxonomy" "term") .Kind) -}} {{- $description = printf "List of all %s used to group the pages of the %s documentation." (lower .Title) .Site.Language.LanguageName -}} {{- end -}} {{- if not $description -}} {{- $summary := .Summary | plainify | htmlUnescape | replaceRE `\s+` " " | replaceRE `^ | $` "" -}} {{- with .Title }}{{ $summary = strings.TrimPrefix (printf "%s " .) $summary }}{{ end -}} {{- $description = $summary | truncate 160 -}} {{- end -}} {{- if not $description -}} {{- $description = .Site.Params.description -}} {{- end -}} {{ with $description }}{{ end }} {{ with .Site.Params.author }}{{ end }} {{- if ne .Kind "404" }} {{ with $description }} {{ end -}} {{- end }}
